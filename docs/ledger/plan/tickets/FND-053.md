## 2. End goal
A pull request that changes Go code fails when a line it changes has a new lint finding. A finding on a line the pull request does not change never fails it. The check is called `go-lint`. It runs on every pull request, for each Go module that the change touches. It also carries the trigger for merge groups, so it runs on every merge group once `main` has a merge queue.

<!-- BEGIN go-lint.yaml -->
```yaml
name: go-lint

# Go lint ratchet (FND-053).
#
# golangci-lint runs on every Go module that a change touches. It reports only
# findings on lines changed since the merge base with main (the setting
# `issues.new-from-merge-base` in /.golangci.yml). An old finding never fails a
# change. A new finding always does. There is no baseline file.
#
# The self-test step proves both halves on a throw-away repository before the
# real run is trusted.
#
# The repository has no root go.mod. Each module is linted from its own
# directory, with the root settings file passed by --config. A settings file
# inside a module is therefore never read.
#
# GO_VERSION must be at least the highest `go` line of any go.mod in the
# repository: golangci-lint refuses a module that is newer than the Go that
# built it. The self-test uses that highest line, so it fails when GO_VERSION
# is too old.

on:
  pull_request:
  merge_group:

permissions:
  contents: read

concurrency:
  group: go-lint-${{ github.ref }}
  cancel-in-progress: true

env:
  GO_VERSION: '1.26'
  GOLANGCI_LINT_VERSION: v2.9.0

jobs:
  go-lint:
    name: go-lint
    runs-on: ubuntu-latest
    timeout-minutes: 45
    steps:
      # Full history: the merge base with origin/main must be reachable.
      - uses: actions/checkout@v4
        with:
          fetch-depth: 0

      # A module is selected when the change touches one of its .go files, its
      # go.mod or its go.sum. A file belongs to the deepest directory above it
      # that holds a go.mod.
      - name: Find the Go modules this change touches
        id: find
        run: |
          set -euo pipefail
          base="$(git merge-base origin/main HEAD)"
          changed="$(git diff --name-only "$base" HEAD)"
          all_modules="$(git ls-files -- 'go.mod' '*/go.mod' | xargs -r -n1 dirname | sort -u)"
          selected=""
          run_lint=false
          while IFS= read -r file; do
            case "$file" in
              .golangci.yml|.github/workflows/go-lint.yaml) run_lint=true; continue ;;
              *.go|*/go.mod|*/go.sum) ;;
              *) continue ;;
            esac
            best=""
            while IFS= read -r dir; do
              case "$file" in
                "$dir"/*) if [ "${#dir}" -gt "${#best}" ]; then best="$dir"; fi ;;
              esac
            done <<< "$all_modules"
            if [ -n "$best" ]; then
              selected="${selected}${best}"$'\n'
              run_lint=true
            fi
          done <<< "$changed"
          selected="$(printf '%s' "$selected" | sort -u)"
          echo "::notice::go-lint: $(printf '%s\n' "$selected" | grep -c . || true) module(s) selected"
          printf '%s\n' "$selected"
          {
            echo "run_lint=${run_lint}"
            echo "modules<<END_OF_MODULES"
            printf '%s\n' "$selected"
            echo "END_OF_MODULES"
          } >> "$GITHUB_OUTPUT"

      - name: Set up Go
        if: steps.find.outputs.run_lint == 'true'
        uses: actions/setup-go@v5
        with:
          go-version: ${{ env.GO_VERSION }}
          cache: false

      - name: Install golangci-lint
        if: steps.find.outputs.run_lint == 'true'
        run: go install "github.com/golangci/golangci-lint/v2/cmd/golangci-lint@${GOLANGCI_LINT_VERSION}"

      - name: Self-test (a new finding fails, an old finding does not)
        if: steps.find.outputs.run_lint == 'true'
        run: |
          set -euo pipefail
          lint="$(go env GOPATH)/bin/golangci-lint"
          newest="$(git ls-files -- 'go.mod' '*/go.mod' | xargs grep -h '^go ' | sort -V | tail -1)"
          T="$(mktemp -d)"
          out="$(mktemp)"
          cd "$T"
          git init -q -b main .
          cp "$GITHUB_WORKSPACE/.golangci.yml" .golangci.yml
          mkdir svc
          printf 'module example.invalid/svc\n\n%s\n' "$newest" > svc/go.mod
          printf 'package svc\n\nimport "os"\n\n// Old holds a finding that exists at the merge base.\nfunc Old() {\n\tos.Remove("old")\n}\n' > svc/old.go
          git add .
          git -c user.name=self-test -c user.email=self-test@example.invalid commit -q -m base
          git update-ref refs/remotes/origin/main HEAD
          # Case 1: the only finding is as old as the merge base. The lint must pass.
          if ! ( cd svc && "$lint" run --config "$T/.golangci.yml" ./... ); then
            echo "::error::go-lint self-test: an old finding failed the lint"
            exit 1
          fi
          # Case 2: a finding is added after the merge base. The lint must fail and name it.
          printf 'package svc\n\nimport "os"\n\n// New holds a finding added after the merge base.\nfunc New() {\n\tos.Remove("new")\n}\n' > svc/new.go
          git add .
          git -c user.name=self-test -c user.email=self-test@example.invalid commit -q -m change
          set +e
          ( cd svc && "$lint" run --config "$T/.golangci.yml" ./... ) > "$out" 2>&1
          rc=$?
          set -e
          cat "$out"
          if [ "$rc" -ne 1 ]; then
            echo "::error::go-lint self-test: a new finding did not fail the lint (exit code ${rc}, want 1)"
            exit 1
          fi
          if ! grep -q 'new\.go' "$out" || ! grep -q 'errcheck' "$out"; then
            echo "::error::go-lint self-test: the new finding was not reported by errcheck"
            exit 1
          fi
          if grep -q 'old\.go' "$out"; then
            echo "::error::go-lint self-test: the old finding was reported"
            exit 1
          fi
          echo "self-test: 2 cases passed"

      - name: Lint each selected module
        if: steps.find.outputs.run_lint == 'true'
        env:
          MODULES: ${{ steps.find.outputs.modules }}
        run: |
          set -euo pipefail
          lint="$(go env GOPATH)/bin/golangci-lint"
          fail=0
          while IFS= read -r module; do
            if [ -z "$module" ]; then continue; fi
            echo "::group::golangci-lint ${module}"
            if ! ( cd "$module" && "$lint" run --config "$GITHUB_WORKSPACE/.golangci.yml" ./... ); then
              echo "::error::go-lint: ${module} has a new finding, or golangci-lint could not run"
              fail=1
            fi
            echo "::endgroup::"
          done <<< "$MODULES"
          exit "$fail"
```
