# PRD-102 — Generated types and server interfaces (oapi-codegen, strict server)

## In plain words

**The problem today.** The open-banking product holds the six interface documents of the standard, but no Go code describes their requests and responses. Every later part of the product would have to write those types by hand, and nothing would show when they drift from the documents.

**What changes.** A code generator of one fixed version, run with go generate, writes one Go package per document: its types and a server interface for the standard library router. The generated code is committed, the product's checks run on it, and a test proves that the generator's settings and its outputs agree.

**Done means:**

- Each of the six interface documents has its own generated Go package, and the product builds and passes go vet with them.
- Running go generate in the generator folder writes the committed packages again byte for byte, so a hand edit turns the existing drift check red.
- The generator is one exact version, pinned in a Go module of its own, and the six documents are unchanged.
- No dependency that the change adds has a known vulnerability: the security scan of the pull request finds nothing new.
- A test of the generator module fails when a document has no generator line, when the settings of one document differ from the others, or when a file the generator did not write lies among the generated ones.
- That test runs on every pull request that changes the product.

The rest of this ticket is the exact instruction for the model that does the work.

## 2. End goal
`products/fingate/internal/gen/` holds six generated Go packages, one per OpenAPI document of profile `rw-4.0.0`: `accountinfo`, `paymentinitiation`, `vrp`, `confirmationfunds`, `events` and `eventnotifications`. Each holds the request and response types of its document and a strict server interface for the standard library router (`net/http`), with one method per operation of the document. The generator is `oapi-codegen` v2.8.2, pinned in the Go module `products/fingate/tools` and run by `go generate ./...` in that folder. Running it again writes the same bytes, which the existing check `go generate changes nothing — products/fingate` proves on every pull request. A test in the generator module proves that the generator lines, the settings files and the generated files agree, and that test runs on every pull request that changes the product. No Go module that the change adds or changes brings a finding that the check `security-scan` counts as new. The six documents are not changed.

// The tests of this file prove that the generator's inputs and outputs fit together:
// one go:generate line per document of profile rw-4.0.0, one settings file per line,
// the same settings in each, and one generated file per document. Whether the generated
// files are up to date is checked by running go generate (the generate job of
// .github/workflows/fingate-ci.yaml), not here.

	"bytes"

const (

var directive = regexp.MustCompile(`^//go:generate go tool oapi-codegen -config oapi/([a-z]+)\.yaml \.\./profiles/rw-4\.0\.0/([a-z-]+-openapi\.yaml)$`)

// settings is the shape of one file oapi/<package>.yaml. A key it does not name is an error.
type settings struct {
	Package       string          `yaml:"package"`
	Output        string          `yaml:"output"`
	Generate      map[string]bool `yaml:"generate"`
	OutputOptions struct {
		Overlay *struct {
			Path string `yaml:"path"`
		} `yaml:"overlay"`
	} `yaml:"output-options"`
}

// check reads the folder products/fingate as fsys and returns one line per broken rule.
func check(fsys fs.FS) []string {
	var fails []string
	fail := func(format string, args ...any) { fails = append(fails, fmt.Sprintf(format, args...)) }

	// Rule 1: every go:generate line has the one allowed form; each package is named once.
	src, err := fs.ReadFile(fsys, "tools/generate.go")
	if err != nil {
		return []string{"rule 1: tools/generate.go cannot be read: " + err.Error()}
	}
	docOf := map[string]string{}
	for _, line := range strings.Split(string(src), "\n") {
		if !strings.HasPrefix(line, "//go:generate") {
			continue
		}
		m := directive.FindStringSubmatch(line)
		if m == nil {
			fail("rule 1: a go:generate line has another form: %s", line)
			continue
		}
		if _, seen := docOf[m[1]]; seen {
			fail("rule 1: package %s is generated twice", m[1])
		}
		docOf[m[1]] = m[2]
	}

	// Rule 2: the lines name each document of the profile exactly once, and the package
	// name is the document name without "-openapi.yaml" and without dashes.
	docs, err := fs.Glob(fsys, profileDir+"/*-openapi.yaml")
	if err != nil || len(docs) == 0 {
		return append(fails, "rule 2: no document found in "+profileDir)
	}
	named := map[string]int{}
	for pkg, doc := range docOf {
		named[doc]++
		if want := strings.ReplaceAll(strings.TrimSuffix(doc, "-openapi.yaml"), "-", ""); pkg != want {
			fail("rule 2: %s is generated as package %s, want %s", doc, pkg, want)
		}
	}
	for _, d := range docs {
		if n := named[path.Base(d)]; n != 1 {
			fail("rule 2: document %s has %d go:generate lines, want 1", path.Base(d), n)
		}
		delete(named, path.Base(d))
	}
	for doc := range named {
		fail("rule 2: a go:generate line names %s, which is not a document of the profile", doc)
	}

	// Rule 3: oapi/ holds one settings file per line, and an overlay only for a file that names it.
	files, err := fs.Glob(fsys, "tools/oapi/*.yaml")
	if err != nil {
		return append(fails, "rule 3: tools/oapi cannot be read: "+err.Error())
	}
	overlays := map[string]bool{}
	for _, f := range files {
		name := path.Base(f)
		if strings.HasSuffix(name, "-overlay.yaml") {
			overlays["oapi/"+name] = false
			continue
		}
		if _, ok := docOf[strings.TrimSuffix(name, ".yaml")]; !ok {
			fail("rule 3: settings file %s is named by no go:generate line", name)
		}
	}

	// Rule 4: each settings file has exactly the shared settings, and rule 5: each
	// generated file exists with the generator's header and its package line.
	pkgs := make([]string, 0, len(docOf))
	for pkg := range docOf {
		pkgs = append(pkgs, pkg)
	}
	sort.Strings(pkgs)
	for _, pkg := range pkgs {
		raw, err := fs.ReadFile(fsys, "tools/oapi/"+pkg+".yaml")
		if err != nil {
			fail("rule 3: settings file %s.yaml is missing", pkg)
			continue
		}
		var s settings
		dec := yaml.NewDecoder(bytes.NewReader(raw))
		dec.KnownFields(true)
		if err := dec.Decode(&s); err != nil {
			fail("rule 4: %s.yaml cannot be read: %v", pkg, err)
			continue
		}
		if s.Package != pkg {
			fail("rule 4: %s.yaml sets package %q, want %q", pkg, s.Package, pkg)
		}
		out := "../internal/gen/" + pkg + "/" + pkg + ".gen.go"
		if s.Output != out {
			fail("rule 4: %s.yaml sets output %q, want %q", pkg, s.Output, out)
		}
		want := map[string]bool{"models": true, "std-http-server": true, "strict-server": true}
		if fmt.Sprint(s.Generate) != fmt.Sprint(want) {
			fail("rule 4: %s.yaml generates %v, want %v", pkg, s.Generate, want)
		}
		if o := s.OutputOptions.Overlay; o != nil {
			if _, ok := overlays[o.Path]; !ok || o.Path != "oapi/"+pkg+"-overlay.yaml" {
				fail("rule 4: %s.yaml names overlay %q, want an existing oapi/%s-overlay.yaml", pkg, o.Path, pkg)
			} else {
				overlays[o.Path] = true
			}
		}
		gen, err := fs.ReadFile(fsys, strings.TrimPrefix(out, "../"))
		if err != nil {
			fail("rule 5: the generated file of %s is missing", pkg)
			continue
		}
		if !bytes.Contains(gen, []byte("\n"+header+"\n")) || !bytes.Contains(gen, []byte("\npackage "+pkg+"\n")) {
			fail("rule 5: internal/gen/%s/%s.gen.go lacks the generator's header or the line package %s", pkg, pkg, pkg)
		}
	}
	for o, used := range overlays {
		if !used {
			fail("rule 3: overlay %s is named by no settings file", o)
		}
	}

	// Rule 5: internal/gen holds the generated files and nothing else.
	all, err := fs.Glob(fsys, "internal/gen/*/*")
	if err != nil {
		return append(fails, "rule 5: internal/gen cannot be read: "+err.Error())
	}
	for _, f := range all {
		pkg := path.Base(path.Dir(f))
		if _, ok := docOf[pkg]; !ok || path.Base(f) != pkg+".gen.go" {
			fail("rule 5: %s is not a generated file of a go:generate line", f)
		}
	}
	sort.Strings(fails)
	return fails
}

			hit := false

			}
			if !hit {
				t.Errorf("want a failure of %s, got %v", strings.TrimSuffix(c.rule, ":"), fails)
			}
		})
	}
}

#!/usr/bin/env python3
"""Count the findings that this change adds, with the scanner of the check security-scan.

Run from the repository root, after the work steps:  python3 /tmp/PRD-102-scan.py

The check security-scan (.github/workflows/security-scan.yaml) scans the merge base and the
change with Trivy and fails when scripts/scan-baseline.py finds a new finding. This script
does the same for the folder products/fingate, the only folder of this change that holds a
dependency file:
  1. It downloads the Trivy release that the workflow pins and checks the workflow's SHA-256.
  2. Probe: a throw-away go.mod with golang.org/x/text v0.3.0 must give at least one finding.
     This proves that the scanner and its database work, so a count of 0 below is measured.
  3. It scans products/fingate as origin/main has it, and as the working tree has it now.
     The scan of the change must read both products/fingate/go.mod and
     products/fingate/tools/go.mod.
  4. It runs scripts/scan-baseline.py compare on the two reports and returns its exit code:
     0 no new finding, 1 a new finding (one line NEW ... each).
Exit code 2: the scan could not be made. Everything is written under /tmp/PRD-102-scan,
which is deleted at the end.
"""
import hashlib
import json
import os
import re
import shutil
import subprocess
import sys
import tarfile
import urllib.request

WORK = "/tmp/PRD-102-scan"
WORKFLOW = ".github/workflows/security-scan.yaml"
FOLDER = "products/fingate"


def fail(msg):
    print("scan: " + msg, flush=True)
    return 2


def main():
    shutil.rmtree(WORK, ignore_errors=True)
    os.makedirs(WORK)
    try:
        return run()
    finally:
        shutil.rmtree(WORK, ignore_errors=True)


def run():
    text = open(WORKFLOW, encoding="utf-8").read()
    version = re.search(r"^  TRIVY_VERSION: (\S+)$", text, re.M)
    digest = re.search(r"^  TRIVY_TARBALL_SHA256: ([0-9a-f]{64})$", text, re.M)
    if not version or not digest:
        return fail(WORKFLOW + " names no Trivy version or SHA-256")
    version, digest = version.group(1), digest.group(1)
    url = ("https://github.com/aquasecurity/trivy/releases/download/"
           f"v{version}/trivy_{version}_Linux-64bit.tar.gz")
    tarball = os.path.join(WORK, "trivy.tar.gz")
    try:
        urllib.request.urlretrieve(url, tarball)
    except OSError as exc:
        return fail(f"cannot download {url}: {exc}")
    with open(tarball, "rb") as fh:
        got = hashlib.sha256(fh.read()).hexdigest()
    if got != digest:
        return fail(f"the Trivy download has SHA-256 {got}, the workflow pins {digest}")
    with tarfile.open(tarball) as tf:
        if hasattr(tarfile, "data_filter"):
            tf.extract("trivy", WORK, filter="data")
        else:
            tf.extract("trivy", WORK)
    os.remove(tarball)
    trivy = os.path.join(WORK, "trivy")
    print(f"scan: Trivy {version}, SHA-256 of the release checked", flush=True)

    def scan(target, report, first=False):
        cmd = [trivy, "fs", "--cache-dir", os.path.join(WORK, "cache"), "--scanners", "vuln,secret",
               "--format", "json", "--timeout", "20m", "--quiet", "--skip-version-check",
               "--output", report, target]
        if not first:
            cmd.insert(-3, "--skip-db-update")
        if subprocess.run(cmd).returncode != 0:
            return None
        with open(report, encoding="utf-8") as fh:
            data = json.load(fh)
        return {r["Target"]: r for r in data.get("Results") or []}

    probe = os.path.join(WORK, "probe")
    os.makedirs(probe)
    with open(os.path.join(probe, "go.mod"), "w", encoding="utf-8") as fh:
        fh.write("module probe.invalid/app\n\ngo 1.22\n\nrequire golang.org/x/text v0.3.0\n")
    found = scan(probe, os.path.join(WORK, "probe.json"), first=True)
    if found is None or not (found.get("go.mod") or {}).get("Vulnerabilities"):
        return fail("probe: the known vulnerable go.mod gave no finding; the scanner does not work")
    print("scan: probe found the known vulnerability", flush=True)

    base = os.path.join(WORK, "base")
    os.makedirs(base)
    archive = os.path.join(WORK, "base.tar")
    with open(archive, "wb") as fh:
        if subprocess.run(["git", "archive", "origin/main", FOLDER], stdout=fh).returncode != 0:
            return fail("git archive origin/main failed")
    with tarfile.open(archive) as tf:
        if hasattr(tarfile, "data_filter"):
            tf.extractall(base, filter="data")
        else:
            tf.extractall(base)
    base_found = scan(os.path.join(base, FOLDER), os.path.join(WORK, "base.json"))
    head_found = scan(FOLDER, os.path.join(WORK, "head.json"))
    if base_found is None or head_found is None:
        return fail("a Trivy scan failed")
    for target in ("go.mod", "tools/go.mod"):
        if target not in head_found:
            return fail(f"the scan of the change did not read {FOLDER}/{target}")
    rc = subprocess.run([sys.executable, "scripts/scan-baseline.py", "compare",
                         os.path.join(WORK, "base.json"), os.path.join(WORK, "head.json")]).returncode
    print(f"scan: exit code of the comparison {rc}")
    return rc


if __name__ == "__main__":
    sys.exit(main())
