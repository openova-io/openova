# FND-164 — Console regression: Organizations, users and access, settings

## In plain words

**The problem today.** The console regression list holds 73 rows about the Organizations directory and detail pages, the Create-organization form, an Organization's life to its deletion, the Users and Settings pages, the identity provider's realm, single sign-on into every application of the Sovereign, the tool server and the customer's sign-in by mailed code, that people walk by hand. The gate check of these surfaces points at the folder of the tests and enters the gate as expected to fail, so every live gate run runs them and records the result; the rows stay in the ledger until a later task deletes them together with the scripts that read them.

+/**
+ * The application of a Pod as the treemap names it (handler/dashboard.go applicationKey): its instance
+ * label, its name label, else the workload that owns it. A Pod owned by a Job is one-shot activity and
+ * has none (isEphemeralRow).
+ */
+export function applicationOf(p: Obj, rsOwner: Map<string, string>): string | null {
+  const owners = p.metadata.ownerReferences ?? []
+  if (owners.some((o) => o.kind === 'Job')) return null
+  const l = p.metadata.labels ?? {}
+  if (l['app.kubernetes.io/instance']) return l['app.kubernetes.io/instance']
+  if (l['app.kubernetes.io/name']) return l['app.kubernetes.io/name']
+  const o = owners.find((x) => x.name !== '')
+  if (!o) return p.metadata.name
+  if (o.kind !== 'ReplicaSet') return o.name
+  return rsOwner.get(`${p.metadata.namespace ?? ''}/${o.name}`) || o.name

+/**

+export async function applicationsIn(namespaces: (r: Region) => Promise<string[]>, regions: readonly Region[] = REGIONS): Promise<{ all: Set<string>; running: Set<string> }> {
+  const all = new Set<string>()
+  const running = new Set<string>()
+  for (const r of regions) {
+    for (const ns of await namespaces(r)) {
+      const rs = new Map((await list(r, 'replicasets', ns)).map((x) => [`${ns}/${x.metadata.name}`, x.metadata.ownerReferences?.[0]?.name ?? '']))
+      for (const p of await list(r, 'pods', ns)) {
+        const a = applicationOf(p, rs)
+        if (!a) continue
+        all.add(a)
+        if (str(p, 'status.phase') === 'Running') running.add(a)
+      }
+    }
+  }
+  return { all, running }

+/** An IMAP session over implicit TLS: one command at a time, the untagged lines and literals it answered. */
+function imap(port: number, servername: string): Promise<{ cmd: (c: string) => Promise<string>; close: () => void }> {
+  return new Promise((ready, fail) => {
+    const sock = connect({ host: '127.0.0.1', port, servername })
+    sock.setEncoding('latin1')
+    let buf = ''
+    let greeted = false
+    let n = 0
+    let wait: { tag: string; verb: string; done: (s: string) => void; fail: (e: Error) => void } | null = null
+    const cmd = (c: string) => new Promise<string>((done, f) => {
+      const tag = `g${++n}`
+      wait = { tag, verb: c.split(' ')[0], done, fail: f }
+      sock.write(`${tag} ${c}\r\n`)
+    })
+    sock.on('data', (d: string) => {
+      buf += d
+      if (!greeted) {
+        if (buf.includes('\r\n')) { greeted = true; buf = ''; ready({ cmd, close: () => sock.end() }) }
+        return
+      }
+      const w = wait
+      if (!w) return
+      const end = new RegExp(`(^|\\r\\n)${w.tag} (OK|NO|BAD)[^\\r\\n]*\\r\\n`).exec(buf)
+      if (!end) return
+      const out = buf
+      buf = ''
+      wait = null
+      if (end[2] === 'OK') w.done(out)
+      else w.fail(new Error(`IMAP ${w.verb} answered ${end[2]}`)) // never the command: LOGIN holds the password
+    })
+    sock.on('error', (e) => (wait ? wait.fail(e) : fail(e)))
+  })
+}

+/** The UIDs of the sign-in mails to an address, and the code of the newest one above a floor. */
+async function codeMails(m: Mailbox, floor: number): Promise<{ newest: number; code: string }> {
+  const fwd = await forward(m)
+  try {
+    const s = await imap(fwd.port, m.host)
+    try {
+      await s.cmd(`LOGIN ${JSON.stringify(m.user)} ${JSON.stringify(m.password)}`)
+      await s.cmd('EXAMINE INBOX')
+      const found = /\* SEARCH([\d ]*)/.exec(await s.cmd(`UID SEARCH TO ${JSON.stringify(m.address)} SUBJECT ${JSON.stringify(SUBJECT)}`))
+      const uids = (found?.[1] ?? '').trim().split(/\s+/).filter((x) => x !== '').map(Number)
+      const newest = Math.max(0, ...uids)
+      if (newest <= floor) return { newest, code: '' }
+      const body = (await s.cmd(`UID FETCH ${newest} BODY.PEEK[TEXT]`)).replace(/=\r\n/g, '')
+      const code = /login code:\s*(\d{6})\b/i.exec(body)?.[1] ?? />\s*(\d{6})\s*</.exec(body)?.[1] ?? ''
+      return { newest, code }
+    } finally {
+      await s.cmd('LOGOUT').catch(() => undefined)
+      s.close()
+    }
+  } finally {
+    fwd.stop()
+  }
+}
