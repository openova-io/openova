# FND-093 — Remove the Sandbox surface of the API

**What changes.** The nine routes, their handlers, their tests and the publisher are deleted. The Organization allowlist no longer names the Sandbox routes, and the two stored size counts of the API server are lowered to match.

**Done means:**

- The API server registers no route under the Sandbox path.
- The handler and event-publisher packages hold no Sandbox or personal-subscription file.
- Every other route still works: the API server builds and all of its tests pass.
- An Organization session is refused on the old Sandbox paths, and a test proves it.
- The list of deleted paths names the nine files, so a pull request that brings one back turns red.

#!/usr/bin/env python3
"""FND-093: remove the Sandbox surface from four files of the API server.

Run from the repository root: python3 /tmp/FND-093-edit.py
Each edit finds its text exactly once and checks how many lines it removes.
If one edit does not fit, the script writes no file and exits with code 1.
"""
import sys

API = "products/catalyst/bootstrap/api/"
MAIN = API + "cmd/api/main.go"
HANDLER = API + "internal/handler/handler.go"
SCOPE = API + "internal/handler/org_scope.go"
SCOPE_TEST = API + "internal/handler/org_scope_test.go"


class Misfit(Exception):
    pass


def once(text, needle, what):
    n = text.count(needle)
    if n != 1:
        raise Misfit("%s: found %d times, expected once" % (what, n))
    return text.index(needle)


def cut(text, first, last, lines, what):
    """Remove from the start of the line that holds `first` to the end of `last` (which ends with a newline)."""
    start = text.rfind("\n", 0, once(text, first, what + " (start)")) + 1
    end = once(text, last, what + " (end)") + len(last)
    if end <= start:
        raise Misfit("%s: the end comes before the start" % what)
    if text[start:end].count("\n") != lines:
        raise Misfit("%s: would remove %d lines, expected %d" % (what, text[start:end].count("\n"), lines))
    return text[:start] + text[end:]


def swap(text, old, new, what):
    once(text, old, what)
    return text.replace(old, new)


def edit_main(t):
    t = cut(t, "Sandbox-create NATS publish. Wires a",
            "\th.SetTenantEventPublisher(newTenantEventPublisherFromEnv(log))\n\n", 10, "main.go publisher wiring")
    t = cut(t, "\t\t// Sandbox BYOS ",
            '\t\trg.Delete("/api/v1/sandbox/sessions/{id}", h.HandleDeleteSandboxSession)\n\n', 34, "main.go Sandbox routes")

            "\t\treturn nil\n\t}\n\treturn pub\n}\n\n", 38, "main.go newTenantEventPublisherFromEnv")
    t = swap(t, "PR #1918 (the same\n\t// dependency that backs sandbox_publisher.go).\n",
             "PR #1918.\n", "main.go comment on sandbox_publisher.go")
    return t


def edit_handler(t):

            "\ttenantEvents TenantEventPublisher\n\n", 12, "handler.go field tenantEvents")
    t = cut(t, "// SetTenantEventPublisher wires the NATS publisher used by the\n",
            "func (h *Handler) TenantEventPublisherForTest() TenantEventPublisher { return h.tenantEvents }\n\n",
            12, "handler.go setter and accessor")
    return t


def edit_scope(t):
    t = cut(t, "//   - /api/v1/sandbox/          ", "(scoped by sub/org)\n", 1, "org_scope.go comment line")
    t = swap(t, '\t"/api/v1/catalog",\n\t"/api/v1/sandbox/",\n\t"/api/v1/sandbox",\n',
