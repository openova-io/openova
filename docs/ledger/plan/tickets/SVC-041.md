```python
#!/usr/bin/env python3
"""SVC-041-edit.py - the Blueprint file edits of work package SVC-041.

Run once from the repository root. It checks every file first and writes only
when every check holds, so it changes all files or none.

  1. 24 files platform/<folder>/blueprint.yaml: line 4 `  name: <folder>`
     becomes `  name: bp-<folder>`.
  2. 30 files platform/<folder>/blueprint.yaml: the line `  visibility: unlisted`
     is added after the one line that starts with `  version:`.
  3. Two new files, platform/openova-flow-server/blueprint.yaml and
     platform/openova-flow-emitter/blueprint.yaml. Their `  version:` line holds
     the version of the sibling chart/Chart.yaml as it is when the script runs.

Exit 0: all files written. Exit 1: a check failed and no file was written.
"""

import os
import re
import sys

RENAME = """alloy cluster-autoscaler-hcloud coraza dragonfly falco grafana harbor hcloud-ccm
kyverno kyverno-policies langfuse litmus loki mimir opentelemetry reloader seaweedfs
sigstore syft-grype tempo trivy velero velero-hcs vpa""".split()

VISIBILITY = """alloy cilium-policies clickhouse coraza debezium external-dns falco ferretdb
flink grafana iceberg kyverno kyverno-policies langfuse litmus loki milvus mimir neo4j
opensearch opentelemetry reflector reloader sigstore strimzi syft-grype tempo trivy
velero velero-hcs""".split()

SERVER = """apiVersion: catalyst.openova.io/v1
kind: Blueprint
metadata:
  name: bp-openova-flow-server
spec:
  version: @VERSION@
  visibility: unlisted
  card:
    title: OpenOva Flow server
  topology:
    supported:
      - singleton
    defaults:
      multi-region: singleton
      single-region: singleton
    perTopology:
      singleton:
        placement:
          tier: mgmt
          clusters: [mgmt-A]
          roles:
            mgmt-A: singleton
"""

EMITTER = """apiVersion: catalyst.openova.io/v1
kind: Blueprint
metadata:
  name: bp-openova-flow-emitter
spec:
  version: @VERSION@
  visibility: unlisted
  card:
    title: OpenOva Flow emitter
  topology:
    supported:
      - singleton
    defaults:
      multi-region: singleton
      single-region: singleton
    perTopology:
      singleton:
        placement:
          tier: ""
          clusters: [mgmt-A, mgmt-B, dmz-A, dmz-B, rtz-A, rtz-B]
          roles:
            mgmt-A: singleton
            mgmt-B: singleton
            dmz-A: singleton
            dmz-B: singleton
            rtz-A: singleton
            rtz-B: singleton
"""

NEW = (("openova-flow-server", SERVER), ("openova-flow-emitter", EMITTER))
SEMVER = re.compile(r"^[0-9]+\.[0-9]+\.[0-9]+$")


def bp(folder):
    return "platform/%s/blueprint.yaml" % folder


def read(path, errors):
    try:
        with open(path, encoding="utf-8") as f:
            return f.read().split("\n")
    except OSError as e:
        errors.append("%s: cannot read (%s)" % (path, e.strerror))
        return None


def plan(errors):
    """The files to write, as {path: text}. Every refusal goes to errors."""
    lines = {}
    for folder in sorted(set(RENAME) | set(VISIBILITY)):
        got = read(bp(folder), errors)
        if got is not None:
            lines[folder] = got
    for folder in RENAME:
        if folder not in lines:
            continue
        ln = lines[folder]
        if len(ln) < 4 or ln[3] != "  name: " + folder:
            errors.append("%s: line 4 is not `  name: %s`" % (bp(folder), folder))
            continue
        ln[3] = "  name: bp-" + folder
    for folder in VISIBILITY:
        if folder not in lines:
            continue
        ln = lines[folder]
        at = [i for i, text in enumerate(ln) if text.startswith("  version:")]
        if len(at) != 1:
            errors.append("%s: %d lines start with `  version:`, expected 1" % (bp(folder), len(at)))
            continue
        if any(text.startswith("  visibility:") for text in ln):
            errors.append("%s: already has a line `  visibility:`" % bp(folder))
            continue
        ln.insert(at[0] + 1, "  visibility: unlisted")
    out = dict((bp(folder), "\n".join(ln)) for folder, ln in lines.items())
    for folder, template in NEW:
        if os.path.lexists(bp(folder)):
            errors.append("%s: exists already" % bp(folder))
            continue
        chart = read("platform/%s/chart/Chart.yaml" % folder, errors)
        if chart is None:
            continue
        names = [text for text in chart if text.startswith("name:")]
        versions = [text.split(":", 1)[1].strip().strip('"').strip("'") for text in chart if text.startswith("version:")]
        if names != ["name: bp-" + folder]:
            errors.append("platform/%s/chart/Chart.yaml: its name line is not `name: bp-%s`" % (folder, folder))
            continue
        if len(versions) != 1 or not SEMVER.match(versions[0]):
            errors.append("platform/%s/chart/Chart.yaml: no single line `version: MAJOR.MINOR.PATCH`" % folder)
            continue
        out[bp(folder)] = template.replace("@VERSION@", versions[0])
    return out


def main():
    if not os.path.isfile("scripts/check-service-model.py"):
        print("SVC-041-edit: run this from the repository root", file=sys.stderr)
        return 1
    errors = []
    out = plan(errors)
    if errors:
        for e in errors:
            print("SVC-041-edit: " + e, file=sys.stderr)
        print("SVC-041-edit: no file written: %d checks failed" % len(errors))
        return 1
    for path in sorted(out):
        with open(path, "w", encoding="utf-8", newline="") as f:
            f.write(out[path])
    print("SVC-041-edit: wrote %d files: %d names, %d visibility lines, %d new files"
          % (len(out), len(RENAME), len(VISIBILITY), len(NEW)))
    return 0


if __name__ == "__main__":
    sys.exit(main())
```
