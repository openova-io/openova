# CTL-355 — Delete the second showback meter

## In plain words

**The problem today.** The platform shows what each Organization uses and costs in two places. The billing engine meters it, prices it and shows it. The console also works out figures of its own, with a different formula and in units that are not money, so the two never agree and nobody can say which one is right.

**What changes.** The console stops working out cost figures of its own. Where it drew them, it shows a link that opens the cost view of the billing engine.

**Done means:**

- The address of the console that returned the second set of cost figures no longer exists.
- On a Sovereign with the billing engine installed, the Organization page, the Organizations directory and the Orders and Revenue pages of a Sovereign that does not sell yet show a link named Open the cost view.
- The link opens the same place as the Chargeback entry of the Billing menu.
- On a Sovereign without the billing engine no link is shown, and the pages still render.
- The treemaps of the dashboard and of the fleet still group workloads by Organization as before.
- The size guard of the API records the deleted code, and the deletion list names the five deleted files.

The rest of this ticket is the exact instruction for the model that does the work.

            with open(path, encoding="utf-8", newline="") as f:

        except OSError as e:
            print("CTL-355 edits: FAIL %s cannot be read: %s" % (path, e))
            bad += 1
            continue
        for n, edit in enumerate(edits, 1):
            count += 1
            if edit[0] == "replace":
                _, old, new = edit
                found = text.count(old)
                print("%s edit %d (replace): old text found %d time(s)" % (path, n, found))
                if found != 1:
                    bad += 1
                    continue
                text = text.replace(old, new)
            elif edit[0] == "cut":
                _, first, second, want = edit
                a, b = text.count(first), text.count(second)
                print("%s edit %d (cut): first text found %d time(s), second text found %d time(s)" % (path, n, a, b))
                if a != 1 or b != 1 or text.index(first) >= text.index(second):
                    bad += 1
                    continue
                start, end = text.index(first), text.index(second)
                got = hashlib.sha256(text[start:end].encode("utf-8")).hexdigest()
                if got != want:
                    print("%s edit %d (cut): the part to remove has SHA-256 %s, expected %s" % (path, n, got, want))
                    bad += 1
                    continue
                text = text[:start] + text[end:]
            else:
                _, marker, tail = edit
                found = text.count(marker)
                print("%s edit %d (append): marker found %d time(s), file ends with a line feed: %s" % (path, n, found, text.endswith("\n")))
                if found != 0 or not text.endswith("\n"):
                    bad += 1
                    continue
                text = text + tail
        out[path] = text
    if bad:
        print("CTL-355 edits: FAIL %d edit(s) could not be made; no file was changed" % bad)
        return 1
    for path, text in out.items():
        with open(path, "w", encoding="utf-8", newline="") as f:
            f.write(text)
    print("CTL-355 edits: %d edits written to %d files" % (count, len(out)))
    return 0

/**
 * CostViewLink — the console's link to the cost view of the billing
 * engine (CTL-355).
 *
 * Usage and cost per Organization and per Application are metered,
 * priced and shown by the billing engine, Catalyst BSS (the Blueprint
 * bp-chargeback), in its own console. This console keeps no meter of its
 * own: where it used to draw a cost table it renders this link.
 *
 * The target is not written here. It is the route of the Billing menu's
 * application section (BILLING_SECTIONS in ../bss/billing-sections.ts),
 * which is the route the Blueprint declares in consoleUI.sidebarRoute.
 * So this link, the Billing menu and the sidebar rail open one surface.
 *
 * The link renders only when the Blueprint registered its console UI on
 * this Sovereign: the presence signal BillingSectionNav uses, read with
 * the same query key so both share one request. While that is not known,
 * or when the billing engine is not installed, nothing is rendered: a
 * link that leads nowhere is worse than no link.
 */

import { Link } from '@tanstack/react-router'
import { useQuery } from '@tanstack/react-query'
import { billingSectionsFor } from '../bss/billing-sections'
import { getSidebarEntries } from '@/lib/console-ui.api'
import { useResolvedDeploymentId } from '@/shared/lib/useResolvedDeploymentId'

export interface CostViewLinkProps {
  /** Classes added to the card, for the spacing its place needs. */
  className?: string
  /** Test seam — the Blueprint ids taken as installed; nothing is fetched. */
  installedOverride?: readonly string[]
}

export function CostViewLink({ className, installedOverride }: CostViewLinkProps) {
  const { deploymentId } = useResolvedDeploymentId()
  const entries = useQuery({
    queryKey: ['billing-nav-sidebar-entries', deploymentId],
    enabled: !installedOverride && !!deploymentId,
    staleTime: 60_000,
    retry: false,
    queryFn: () => getSidebarEntries(deploymentId as string),
  })
  const installed =
    installedOverride ?? (entries.data ?? []).filter((e) => e.enabled).map((e) => e.id)
  const section = billingSectionsFor(installed).find((s) => s.id === 'chargeback')
  if (!section) return null

  return (
    <section
      data-testid="cost-view-link"
      className={`rounded-xl border border-[var(--color-border)] bg-[var(--color-surface)] p-4 ${className ?? ''}`.trim()}
    >
      <h2 className="mb-1 text-sm font-semibold text-[var(--color-text-strong)]">Usage and cost</h2>
      <p className="mb-3 text-sm text-[var(--color-text-dim)]">
        Usage and cost per Organization and per Application are metered and priced in
        Catalyst BSS.
      </p>
      <Link
        to={section.to as never}
        data-testid="cost-view-link-open"
        className="rounded-md border border-[var(--color-border)] bg-[var(--color-surface)] px-2.5 py-1 text-xs font-medium text-[var(--color-accent)] no-underline transition-colors hover:border-[var(--color-accent)]"
      >
        Open the cost view
      </Link>
    </section>
  )
}
