    for old, new, times in pairs:
        count = content.count(old)
        if count != times:
            first = old.strip("\n").splitlines()[0].strip()
            errors.append("%s: the text %r occurs %d times, not %d" % (path, first, count, times))
            continue
        content = content.replace(old, new)
    out[path] = content

for path, pairs in LOWER_YAML.items():
    content = read(path)
    if content is None:
        continue
    for key, by in pairs:
        pattern = re.compile(r"^  " + re.escape(key) + r": ([0-9]+)$", re.M)
        hits = pattern.findall(content)
        if len(hits) != 1 or int(hits[0]) < by:
            errors.append("%s: the count line of %s occurs %d times, or is below %d" % (path, key, len(hits), by))
            continue
        content = pattern.sub("  %s: %d" % (key, int(hits[0]) - by), content)
    out[path] = content

for path, pairs in LOWER_EQ.items():
    content = read(path)
    if content is None:
        continue
    for key, by in pairs:
        pattern = re.compile(r"^" + re.escape(key) + r"=([0-9]+)$", re.M)
        hits = pattern.findall(content)
        if len(hits) != 1 or int(hits[0]) < by:
            errors.append("%s: the line of %s occurs %d times, or is below %d" % (path, key, len(hits), by))
            continue
        content = pattern.sub("%s=%d" % (key, int(hits[0]) - by), content)
    out[path] = content

content = read(DELETED_LIST)
if content is not None:
    if not content.endswith("\n"):
        errors.append("%s: the last line has no line end" % DELETED_LIST)
    elif FOLDER in content.split("\n"):
        errors.append("%s: the folder is listed already" % DELETED_LIST)
    else:
        out[DELETED_LIST] = content + FOLDER + "\n"

if errors:
    for error in errors:
        print("SEC-105-edit: nothing changed: " + error)
    sys.exit(1)
for name in sorted(FILES):
    os.remove(os.path.join(FOLDER, name))
    print("SEC-105-edit: deleted " + FOLDER + "/" + name)
for top, _dirs, _names in sorted(os.walk(FOLDER), key=lambda entry: -len(entry[0])):
    os.rmdir(top)
