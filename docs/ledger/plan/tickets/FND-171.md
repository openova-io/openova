"""FND-171: remove the DR runbook playback and preflight routes from the API server.

Run from the repository root: python3 /tmp/FND-171-edit.py
Each edit finds its text exactly once and checks how many lines it removes.
If one edit does not fit, the script writes no file and exits with code 1.
"""
import sys

API = "products/catalyst/bootstrap/api/"
MAIN = API + "cmd/api/main.go"
DR = API + "internal/handler/continuum_dr_extras.go"


class Misfit(Exception):
    pass


def once(text, needle, what):
    n = text.count(needle)
    if n != 1:
        raise Misfit("%s: found %d times, expected once" % (what, n))
    return text.index(needle)


def cut(text, first, last, lines, what):
    """Remove from the start of the line that holds `first` to the end of `last` (which ends with a newline)."""
    start = text.rfind("\n", 0, once(text, first, what + " (start)")) + 1
    end = once(text, last, what + " (end)") + len(last)
    if end <= start:
        raise Misfit("%s: the end comes before the start" % what)
    if text[start:end].count("\n") != lines:
        raise Misfit("%s: would remove %d lines, expected %d" % (what, text[start:end].count("\n"), lines))
    return text[:start] + text[end:]


def swap(text, old, new, what):
    once(text, old, what)
    return text.replace(old, new)


def edit_main(t):
    t = swap(t, '\t\trg.Post("/api/v1/sovereigns/{id}/dr/runbook/preflight", h.HandleDRRunbookPreflight)\n'
                '\t\trg.Post("/api/v1/sovereigns/{id}/dr/runbook/playback", h.HandleDRRunbookPlayback)\n',
             "", "main.go route lines")
    t = swap(t, "\t\t// replication-status, switchover-history, runbook preflight +\n"
                "\t\t// playback, quorum status, sovereign-wide replication roll-up,\n",
             "\t\t// replication-status, switchover-history, quorum status,\n"
             "\t\t// sovereign-wide replication roll-up,\n", "main.go comment on the route list")
    t = swap(t, "\t\t// Per INVIOLABLE-PRINCIPLES #5 the playback POST + settings PUT\n"
                "\t\t// gate on owner tier (REUSE applicationInstallCallerAuthorized);\n"
                "\t\t// preflight + GET endpoints gate on viewer.\n",
             "\t\t// The GET endpoints gate on viewer.\n", "main.go comment on the gates")
    return t


def edit_dr(t):
    t = swap(t, "//\tPOST   /api/v1/sovereigns/{id}/dr/runbook/preflight\n"
                "//\tPOST   /api/v1/sovereigns/{id}/dr/runbook/playback\n",
             "", "header route lines")
    t = swap(t, '\t"context"\n\t"encoding/json"\n\t"fmt"\n', '\t"context"\n\t"fmt"\n',
             "import of encoding/json, which only the playback handler used")
    t = swap(t, "// The runbook + quorum endpoints surface the DR runbook's preflight gates\n"
                "// (10-step health check) and the playback step (executes the preflight,\n"
                "// records the result on the audit trail, and emits the\n"
                "// `dr-runbook-playback` event on NATS). Per ADR-0001 §2.7 every CR\n",
             "// The quorum endpoint surfaces the DNS-quorum witness state the\n"
             "// Continuum controller reports. Per ADR-0001 §2.7 every CR\n",
             "header paragraph on the runbook endpoints")
    t = swap(t, "// hostnames). Per #5 the playback POST gates on owner tier (REUSES\n"
                "// applicationInstallCallerAuthorized — same gate as switchover); the\n"
                "// preflight + read endpoints gate on viewer (any authenticated tier).\n",
             "// hostnames). Per #5 the read endpoints gate on viewer (any\n"
             "// authenticated tier).\n",
             "header sentence on the gates")
    t = cut(t, "// drRunbookPreflightCheck — one row in the DR runbook preflight matrix.\n",
            "\tDurationMs int64  `json:\"durationMillis\"`\n\tMessage    string `json:\"message,omitempty\"`\n}\n\n",
            58, "types of the two routes")
    t = cut(t, "// HandleDRRunbookPreflight — POST\n",
            "\t\tresp.AuditEventID = fmt.Sprintf(\"audit-%s-%d\", contName, completedAt.UnixNano())\n"
            "\t}\n\twriteJSON(w, http.StatusOK, resp)\n}\n\n",
            135, "handlers of the two routes")
    t = cut(t, "// Preflight check statuses — the vocabulary drRunbookPreflightCheck.Status\n",
            "\treturn schema.GroupVersionResource{Group: \"dr.openova.io\", Version: \"v1\", Resource: \"pdms\"}\n}\n\n",
            26, "preflight statuses and the two kinds")
    t = cut(t, "// numericNestedFound reads a numeric field AND reports whether it was present.\n",
            "\tdefault:\n\t\treturn \"Ready\", blocking\n\t}\n}\n\n",
            451, "preflight helpers")
    return t


EDITS = [(MAIN, edit_main), (DR, edit_dr)]


def main():
